def is_valid_amount(amount):
    return amount > 0 and amount <= 100000

def is_valid_pin(pin):
    return pin == saved_pin

def is_valid_upi_id(upi_id):
    parts = upi_id.split("@")
    return len(parts) == 2 and len(parts[0]) > 0 and len(parts[1]) > 0

def mask(card):
    return "**** **** **** " + card[-4:]

def fee_for(mode):
    if mode == "card":
        return 5
    elif mode == "netbanking":
        return 2
    return 0

def cashback_for(amount):
    if amount >= 5000:
        return 100
    elif amount >= 2000:
        return 50
    elif amount >= 500:
        return 10
    return 0

def mode_name(mode):
    match mode:
        case "upi":
            return "UPI"
        case "card":
            return "Card"
        case "netbanking":
            return "Net banking"
        case _:
            return "Unknown mode"

def print_receipt(name, amount, mode, fee, cashback, balance):
    print(f"Paid ₹{amount} to {name} via {mode_name(mode)}")
    print(f"Fee ₹{fee} | Cashback ₹{cashback} | Balance ₹{balance}")

def ask_pin():
    for attempt in range(3):
        pin = input("UPI PIN: ").strip()
        if is_valid_pin(pin):
            return True
        print(f"Wrong PIN. {2 - attempt} attempts left")
    return False

def pay(balance, history, upi_id, amount, mode="upi"):
    if not is_valid_upi_id(upi_id):
        print(f"'{upi_id}' is not a valid UPI ID. Payment refused.")
        return balance
    if not is_valid_amount(amount):
        print("Invalid amount. Payment refused.")
        return balance
    if mode_name(mode) == "Unknown mode":
        print(f"Unknown payment mode '{mode}'. Payment refused.")
        return balance
    fee = fee_for(mode)
    if amount + fee > balance:
        print("Insufficient balance. Payment refused.")
        return balance
    cashback = cashback_for(amount)
    balance = balance - amount - fee + cashback
    name = upi_id.split("@")[0].title()
    print_receipt(name, amount, mode, fee, cashback, balance)
    history.append(amount)
    return balance

        total += amount

Why does the set order change between runs?

Real incident · 2011

One web request, hours of CPU

  • Python, PHP, Java and Ruby all stored sets and dictionaries by hashing each key.
  • Researchers crafted thousands of keys that hashed to the same spot. Lookups slowed from instant to crawling.
  • One HTTP request with those keys as form fields could pin a server's CPU at 100% for hours.
Python's fix, in 3.3: a random hash seed on every run. That is why your set prints in a different order each time. The order was never a promise.
Read: oCERT advisory 2011-003 · Python docs: hash randomization

Which line fails?

b = {["B", 12]: "Rajani"}   # B
c = {"B": [12, 13]}         # C
B. TypeError: unhashable type: 'list'. A key must be something that cannot change: a string, a number, a tuple. A list can change, so it cannot be a key. A list as a value is fine, so C works.
